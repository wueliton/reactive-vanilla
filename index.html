The DOM is already a rendering engine.

A tiny reactive runtime for building modern interfaces with vanilla JavaScript — directly on the DOM.

Get started Explore examples 
component('.counter', () => {
    const state = signal({
        count: 1
    });

    el('.counter-display', {
        textContent: () => state.count
    });

    el('.increment', {
        onclick: () => state.count++
    });

    el('.decrement', {
        onclick: () => state.count--
    });
});
<div class="counter">
  <button class="decrement">-</button>
  <div class="counter-display">1</div>
  <button class="increment">+</button>
</div>
